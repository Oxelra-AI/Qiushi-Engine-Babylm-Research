% Label-free model symbol: a repeated block with residual bypasses.
% Broad slabs and thin bars abstract the two operations and their merges.
% No operator names, acronyms, equations, or layer counts belong in this icon.
% It identifies a Transformer-family model, not an exact architecture.
% All coordinates are millimetres at the figure's final publication size.
\newcommand{\ResearchModelGlyph}[3]{%
  \begin{scope}[shift={(#1,#2)},
    draw=#3,line width=.45bp,line cap=round,line join=round,
    flow/.style={-{Latex[length=.60mm,width=.48mm]},line width=.40bp}]
    \foreach \d in {.75,0}{
      \draw[fill=white,draw=#3!55,line width=.35bp,rounded corners=.35mm]
        (-5.8+\d,-5.9+\d) rectangle (5.8+\d,5.9+\d);
    }
    \draw (0,-5.25) -- (0,-4.7);
    \draw[flow] (0,-2.7) -- (0,-1.15);
    \draw[flow] (0,-.45) -- (0,.4);
    \draw[flow] (0,2.4) -- (0,3.95);
    \draw (0,4.65) -- (0,5.35);
    \draw[flow] (0,-5.25) -- (-5.10,-5.25) -- (-5.10,-.8) -- (-4.4,-.8);
    \draw[flow] (0,-.03) -- (-5.10,-.03) -- (-5.10,4.3) -- (-4.4,4.3);
    \foreach \y in {-3.7,1.4}{
      \draw[fill=#3!12,rounded corners=.22mm]
        (-4.4,\y-1.0) rectangle (4.4,\y+1.0);
    }
    \foreach \y in {-.8,4.3}{
      \draw[fill=white,rounded corners=.12mm]
        (-4.4,\y-.35) rectangle (4.4,\y+.35);
    }
  \end{scope}%
}
